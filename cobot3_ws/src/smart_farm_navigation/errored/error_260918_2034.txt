cd /home/rokey/cobot3_ws
ros_set
colcon build --packages-select smart_farm_navigation --symlink-install
source install/setup.bash
ros2 run smart_farm_navigation escape_controller --ros-args --params-file /home/rokey/cobot3_ws/src/smart_farm_navigation/config/escape_controller.yaml
